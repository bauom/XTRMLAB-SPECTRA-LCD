# hongtai_screen -- everything needed to run the desktop app with all
# four themes available. Windows only.
#
# The shipped desktop app is now scripts/run_v2_app.py (the backend +
# webview window showing the React frontend) -- that's what the
# desktop icon and "Launch at Windows startup" point at. app.py's
# Tkinter GUI still works (`python app.py`) for manual/headless use,
# it's just not what gets launched automatically any more.
#
# Only want one theme instead of the full desktop app? You don't need
# all of this -- see that theme's own section in README.md for just
# its dependencies.
#
#   pip install -r requirements.txt

# --- Core: the driver itself + the desktop app ------------------------
pyserial>=3.5
pillow>=10.0          # ImageOps is used for the Dashboard's custom-image background

# --- demo_clock.py: CPU/RAM bars ---------------------------------------
psutil>=5.9

# --- dashboard_theme.py: gauges, CPU/GPU stats, Spotify art, weather --
# weather.py (the dashboard's optional weather middle-content option)
# uses only stdlib urllib -- no extra package needed, just a working
# internet connection to reach Open-Meteo.
numpy>=1.24          # required -- gauge rendering + background gradients
pycairo>=1.24        # required -- gauge rendering (prebuilt wheel on Windows)
nvidia-ml-py>=12.0   # optional -- NVIDIA GPU/VRAM/power stats; degrades gracefully without one
winsdk>=1.0.4        # optional -- Spotify/now-playing album art (Windows 10 1809+)
pycaw>=20240210      # optional -- the "Volume" stat (Windows master output level); without it
                     #             that one stat reads "--" and nothing else is affected

# --- video_theme.py: stream a video file to the panel ------------------
opencv-python-headless>=4.8
# pygame            # optional -- only needed for --audio; also needs ffmpeg on PATH

# --- webpage_theme.py: mirror a webpage to the panel --------------------
playwright>=1.40
# after installing, run once:  playwright install chromium   (~150MB download)

# --- Desktop app extra: system tray icon on close/at startup -----------
pystray>=0.19        # optional -- without it, closing the window quits instead
                     # of dismissing to tray, and --autostart just minimizes

# --- Desktop app window: scripts/run_v2_app.py / scripts/run_ui.py ----
pywebview>=5.0        # required -- this is the actual app window now (the
                     # desktop icon/"Launch at Windows startup" both point
                     # at run_v2_app.py). Also needs WebView2 (ships with
                     # Windows 11; usually present on Windows 10 via Edge).

# --- Building a standalone .exe (developers only, not needed to run) --
# pyinstaller>=6.0  # see BUILD.md -- uncomment or `pip install pyinstaller` yourself
